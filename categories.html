document.addEventListener("DOMContentLoaded", async () => { const searchInput = document.getElementById("searchInput"); const genreFilter = document.getElementById("genreFilter"); const animeListEl = document.getElementById("animeList"); const recommendationsBox = document.getElementById("recommendationsBox"); if (!animeListEl) return; let allAnimes = []; function getStars(rating) { const fullStars = Math.floor(rating); const hasHalfStar = rating % 1 >= 0.5; let stars = "★".repeat(fullStars); if (hasHalfStar) stars += "½"; return stars; } function renderRecommendations() { if (!recommendationsBox) return; const watchlistMap = JSON.parse(localStorage.getItem("animeFlix_watchlist") || "{}"); const selectedIds = Object.entries(watchlistMap) .filter(([, status]) => status && status !== "watchlist") .map(([id]) => Number(id)); if (!selectedIds.length) { recommendationsBox.innerHTML = `

Recommandations

Ajoutez quelques titres dans votre liste pour obtenir des suggestions.
`; return; } const genres = []; selectedIds.forEach((id) => { const anime = allAnimes.find((item) => Number(item.id) === id); if (anime) genres.push(...(anime.genres || [])); }); const recommended = allAnimes .filter((anime) => !selectedIds.includes(Number(anime.id))) .map((anime) => { const score = (anime.genres || []).filter((genre) => genres.includes(genre)).length; return { anime, score }; }) .sort((a, b) => b.score - a.score || b.anime.rating - a.anime.rating) .slice(0, 3); recommendationsBox.innerHTML = `

Recommandations

Basées sur vos préférences
${recommended.map(({ anime }) => `
${anime.title} ${(anime.genres || []).slice(0, 2).join(" • ")}
`).join("")}
`; } function renderCards(list) { const favorites = JSON.parse(localStorage.getItem("animeFlix_favorites") || "{}"); const watchlistMap = JSON.parse(localStorage.getItem("animeFlix_watchlist") || "{}"); animeListEl.innerHTML = list.map((anime) => { const favorite = Boolean(favorites[String(anime.id)]); const watchlistStatus = watchlistMap[String(anime.id)] || "watchlist"; const label = { watchlist: "À regarder", watching: "En cours", completed: "Terminé" }[watchlistStatus] || "À regarder"; return `

${anime.title}

⭐ ${anime.rating}/5 ${anime.releaseYear}

${anime.description}

${(anime.genres || []).map((g) => `${g}`).join("")}
Voir détails
${label}
`; }).join(""); animeListEl.querySelectorAll(".favorite-toggle").forEach((button) => { button.addEventListener("click", () => { const id = Number(button.dataset.id); const isActive = window.AnimeFlix.toggleFavorite(id); button.classList.toggle("active", isActive); button.textContent = isActive ? "♥" : "♡"; }); }); animeListEl.querySelectorAll(".watchlist-select").forEach((select) => { select.addEventListener("change", (event) => { const id = Number(event.target.dataset.id); const status = event.target.value; window.AnimeFlix.setWatchlistStatus(id, status); renderRecommendations(); renderCards(allAnimes); }); }); } function applyFilters() { const query = (searchInput?.value || "").trim().toLowerCase(); const selectedGenre = genreFilter?.value || "Tous"; const filtered = allAnimes.filter((anime) => { const haystack = [ anime.title, anime.author, anime.studio, anime.description, (anime.genres || []).join(" ") ].join(" ").toLowerCase(); const matchQuery = !query || haystack.includes(query); const matchGenre = selectedGenre === "Tous" || (anime.genres || []).includes(selectedGenre); return matchQuery && matchGenre; }); renderCards(filtered); } allAnimes = await window.AnimeFlix.loadAnimeCatalog(); const genres = [...new Set(allAnimes.flatMap((anime) => anime.genres || []))].sort(); if (genreFilter) { genreFilter.innerHTML = ["Tous", ...genres].map((genre) => ` `).join(""); } if (searchInput) searchInput.addEventListener("input", applyFilters); if (genreFilter) genreFilter.addEventListener("change", applyFilters); renderRecommendations(); applyFilters(); });
