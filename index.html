(function () { const STORAGE_KEYS = { FAVORITES: "animeFlix_favorites", WATCHLIST: "animeFlix_watchlist", RESUME: "animeFlix_resume", ACCOUNT: "animeFlix_account", ADMIN: "animeFlix_admin_animes", SESSIONS: "animeFlix_sessions" }; function safeJsonParse(value, fallback) { try { return value ? JSON.parse(value) : fallback; } catch (error) { return fallback; } } function readStorage(key, fallback) { return safeJsonParse(localStorage.getItem(key), fallback); } function writeStorage(key, value) { localStorage.setItem(key, JSON.stringify(value)); } function normalizeAnime(anime, index) { const seasons = Array.isArray(anime.seasons) && anime.seasons.length ? anime.seasons : [{ number: 1, title: "Saison 1", episodes: [{ number: 1, title: "Épisode 1", duration: "24:00", videoUrl: "" }] }]; return { id: Number(anime.id ?? index + 1), title: anime.title || `Anime ${index + 1}`, rating: Number(anime.rating ?? 4), poster: anime.poster || "images/default-poster.jpg", description: anime.description || "Description à venir.", genres: Array.isArray(anime.genres) ? anime.genres : [], releaseYear: Number(anime.releaseYear ?? 2024), totalEpisodes: Number(anime.totalEpisodes ?? (seasons[0]?.episodes?.length ?? 0)), studio: anime.studio || "Studio principal", author: anime.author || "Auteur inconnu", seasons }; } async function loadAnimeCatalog() { try { const response = await fetch("anime.json"); const data = await response.json(); const defaultAnimes = Array.isArray(data?.animes) ? data.animes : []; const customAnimes = readStorage(STORAGE_KEYS.ADMIN, []); const list = customAnimes.length ? customAnimes : defaultAnimes; return list.map(normalizeAnime); } catch (error) { console.error("Erreur chargement anime.json :", error); return []; } } function getWatchlistStatus(animeId) { const map = readStorage(STORAGE_KEYS.WATCHLIST, {}); return map[String(animeId)] || "watchlist"; } function setWatchlistStatus(animeId, status) { const map = readStorage(STORAGE_KEYS.WATCHLIST, {}); map[String(animeId)] = status; writeStorage(STORAGE_KEYS.WATCHLIST, map); } function isFavorite(animeId) { const favorites = readStorage(STORAGE_KEYS.FAVORITES, {}); return Boolean(favorites[String(animeId)]); } function toggleFavorite(animeId) { const favorites = readStorage(STORAGE_KEYS.FAVORITES, {}); const key = String(animeId); favorites[key] = !Boolean(favorites[key]); writeStorage(STORAGE_KEYS.FAVORITES, favorites); return favorites[key]; } function getResumeState(animeId, seasonNumber, episodeNumber) { const map = readStorage(STORAGE_KEYS.RESUME, {}); const key = `${animeId}:${seasonNumber}:${episodeNumber}`; return map[key] || null; } function saveResumeState(animeId, seasonNumber, episodeNumber, currentTime, duration = 0) { const map = readStorage(STORAGE_KEYS.RESUME, {}); const key = `${animeId}:${seasonNumber}:${episodeNumber}`; map[key] = { animeId, seasonNumber, episodeNumber, currentTime, duration, updatedAt: new Date().toISOString() }; writeStorage(STORAGE_KEYS.RESUME, map); } function getRecommendationsForAnime(currentAnimeId, animeList = []) { const watchlistMap = readStorage(STORAGE_KEYS.WATCHLIST, {}); const selected = Object.entries(watchlistMap) .filter(([, status]) => status && status !== "watchlist") .map(([id]) => Number(id)); const baseAnime = animeList.find((anime) => Number(anime.id) === Number(currentAnimeId)); const baseGenres = baseAnime?.genres?.length ? baseAnime.genres : ["Action", "Drame"]; return animeList .filter((anime) => Number(anime.id) !== Number(currentAnimeId)) .map((anime) => { const score = (anime.genres || []).filter((genre) => baseGenres.includes(genre)).length; return { anime, score }; }) .sort((a, b) => b.score - a.score || b.anime.rating - a.anime.rating) .slice(0, 3) .map((entry) => entry.anime); } function getUserProfile() { return readStorage(STORAGE_KEYS.ACCOUNT, { email: "", password: "", twoFactorEnabled: false, twoFactorSecret: "AFX-2026", recoveryCodes: [], connectedSessions: [] }); } function setUserProfile(profile) { writeStorage(STORAGE_KEYS.ACCOUNT, profile); } function generateRecoveryCodes() { return Array.from({ length: 4 }, () => { const randomNumber = Math.floor(Math.random() * 9000) + 1000; return `AFX-${randomNumber}`; }); } function getConnectedSessions() { return readStorage(STORAGE_KEYS.SESSIONS, []); } function addConnectedSession(session) { const sessions = getConnectedSessions(); const next = [session, ...sessions].slice(0, 5); writeStorage(STORAGE_KEYS.SESSIONS, next); return next; } window.AnimeFlix = { STORAGE_KEYS, loadAnimeCatalog, getWatchlistStatus, setWatchlistStatus, isFavorite, toggleFavorite, getResumeState, saveResumeState, getRecommendationsForAnime, getUserProfile, setUserProfile, generateRecoveryCodes, getConnectedSessions, addConnectedSession }; })();
