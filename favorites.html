if (!values.length) return 0; return values.reduce((sum, val) => sum + val, 0) / values.length; } function formatDurationLabel(minutes) { if (!minutes) return "Durée inconnue"; const hours = Math.floor(minutes / 60); } function matchesCategory(anime, category) { if (!category || category === "Tous") return true; return (anime.genres || []).some( (genre) => genre.toLowerCase() === category.toLowerCase() ); } function matchesDuration(anime, filter) { if (filter === "Tous") return true;
